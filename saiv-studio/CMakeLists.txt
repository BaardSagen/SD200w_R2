cmake_minimum_required(VERSION 3.20)
project(SAIVStudio VERSION 0.3.1 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_source_files_properties(vendor/gsw/gsw_oceanographic_toolbox.c vendor/gsw/gsw_saar.c PROPERTIES LANGUAGE CXX)
add_library(gsw STATIC vendor/gsw/gsw_oceanographic_toolbox.c vendor/gsw/gsw_saar.c)
target_include_directories(gsw PUBLIC vendor/gsw)
add_library(studio_core STATIC src/core.cpp src/sit_compat.cpp)
target_include_directories(studio_core PUBLIC src)
target_link_libraries(studio_core PUBLIC gsw)
set(IMGUI vendor/imgui)
add_executable(SAIVStudio WIN32 src/main.cpp src/app.cpp src/studio_ui.cpp src/plot.cpp src/native_dialogs.cpp src/device_settings.cpp src/serial.cpp src/studio.rc
 ${IMGUI}/imgui.cpp ${IMGUI}/imgui_draw.cpp ${IMGUI}/imgui_tables.cpp ${IMGUI}/imgui_widgets.cpp
 ${IMGUI}/backends/imgui_impl_win32.cpp ${IMGUI}/backends/imgui_impl_dx11.cpp)
target_include_directories(SAIVStudio PRIVATE ${IMGUI} ${IMGUI}/backends assets)
target_compile_definitions(SAIVStudio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(SAIVStudio PRIVATE studio_core d3d11 dxgi d3dcompiler dwmapi setupapi comdlg32 comctl32 shell32 ole32 advapi32)
if(MSVC)
 target_link_options(SAIVStudio PRIVATE /MANIFEST:NO)
 target_compile_options(studio_core PRIVATE /W4 /utf-8)
 target_compile_options(SAIVStudio PRIVATE /W4 /utf-8)
 target_compile_definitions(gsw PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
include(CTest)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE studio_core)
add_test(NAME core_tests COMMAND core_tests ${CMAKE_CURRENT_SOURCE_DIR}/samples)
add_executable(compat_tests tests/compat_tests.cpp)
target_link_libraries(compat_tests PRIVATE studio_core)
add_test(NAME compat_tests COMMAND compat_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures ${CMAKE_CURRENT_BINARY_DIR}/compat-output)
add_executable(settings_tests tests/settings_tests.cpp)
target_include_directories(settings_tests PRIVATE src)
add_test(NAME settings_tests COMMAND settings_tests)
install(TARGETS SAIVStudio RUNTIME DESTINATION .)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY samples docs DESTINATION .)
install(FILES vendor/imgui/LICENSE.txt DESTINATION licenses RENAME ImGui.txt)
install(FILES vendor/gsw/LICENSE DESTINATION licenses RENAME GSW.txt)
